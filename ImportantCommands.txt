===============================================================================
          LOAD BALANCER L4 - COMANDOS DE CONTROLO PARA A DEFESA
===============================================================================

--- 1. PREPARAÇÃO DO AMBIENTE (DOCKER) ---
# Compilar imagens e subir os nós de backend + Redis em segundo plano (detached)
docker-compose up --build -d

# Validar se todos os contentores estão ativos e verificar mapeamento de portas
docker-compose ps

--- 2. COMPILAÇÃO E EXECUÇÃO DO MOTOR (RUST) ---
# Limpar resíduos de compilações anteriores
cargo clean

# Compilar e executar o Load Balancer no MODO RELEASE (Alta Performance Ativada)
cargo run --release

--- 3. MONITORIZAÇÃO COMPUTAÇÃO EM REAL-TIME ---
# Abrir num terminal secundário para provar o isolamento de hardware (cgroups)
docker stats

--- 4. SIMULAÇÃO DE CARGA E STRESS TESTING ---
# Executar o cliente de teste com parâmetros rigorosos:
# Parâmetros: <DURAÇÃO_SEGUNDOS> <CONEXÕES_CONCORRENTES> <TIMEOUT_MS>
# Exemplo base para demonstração estável (500 conexões):
cargo run --release --bin test_client -- 300 500 200

--- 5. INTERAÇÃO DINÂMICA VIA REDIS (HOT-RELOAD UI) ---
# Aceder à linha de comandos interativa do contentor Redis
docker exec -it loadbalancerl4-redis-1 redis-cli

# Comandos Úteis dentro do Redis CLI:
# -> Alterar algoritmo para Weighted Round-Robin:
SET lb:algorithm wrr
# -> Alterar peso do backend1 para capacidade máxima:
SET weight:127.0.0.1:8081 10
# -> Retirar o backend2 da rotação do tráfego (Peso 0):
SET weight:127.0.0.1:8082 0

--- 6. PROCEDIMENTOS DE LIMPEZA E RESET EM CASO DE FALHA ---
# Derrubar a infraestrutura eliminando volumes e dados efêmeros do Redis
docker-compose down -v --remove-orphans

# Forçar o encerramento de processos zombies que prendam os portos críticos
lsof -ti tcp:8080,3000,6379,8081,8082,8083,8084 | xargs kill -9
